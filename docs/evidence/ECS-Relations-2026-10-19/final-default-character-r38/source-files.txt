   9:     EntityPairsWithin source, TRelation relation) : IQueryRelationEnumerable<(Entity, Entity)>
  10:     where TRelation : struct, IRelation<TFirst, TSecond>
  11:     where TFirst : struct
  12:     where TSecond : struct

  19:     public static bool ReadsFromSingleQuery => true;
  20:     public QueryHandle GetQuery(int position) => source.GetQuery(position);
  21:     public QueryDefinition AddReads(QueryDefinition definition, int position)
  22:     {
  23:         QueryDefinition reads = definition.WithOptionalRead<TFirst>();
  24:         return typeof(TFirst) == typeof(TSecond) ? reads : reads.WithOptionalRead<TSecond>();
  25:     }
  26:
  27:     [MethodImpl(MethodImplOptions.AggressiveInlining)]
  28:     public bool Enumerate<TOutput>(scoped RelationQueryViews queries, ref TOutput output)
  29:         where TOutput : struct, IQueryRelationOutput<(Entity, Entity)>, allows ref struct
  30:     {
  31:         QueryCursor members = queries.First;
  32:         return _componentsRequired ? Enumerate<RequiredRelationComponents, TOutput>(members, relation, ref output)
  33:             : Enumerate<OptionalRelationComponents, TOutput>(members, relation, ref output);
  34:     }
  35:
  36:     private static bool ReadComponentRequirements(EntityPairsWithin source)
  37:     {
  38:         QueryDefinition definition = source.World.GetQueryDefinition(source.GetQuery(0));
  39:         return definition.Requires<TFirst>() &&
  40:             (typeof(TFirst) == typeof(TSecond) || definition.Requires<TSecond>());
  41:     }
  42:
  43:     private static bool Enumerate<TPresence, TOutput>(scoped QueryCursor members, TRelation predicate, ref TOutput output)
  44:         where TPresence : struct, IRelationComponentPresence
  45:         where TOutput : struct, IQueryRelationOutput<(Entity, Entity)>, allows ref struct
  46:     {
  47:         var firstRows = members.ReadRows<TFirst, TSecond>();
  48:         while (firstRows.MoveNext(out var first, out var firstValue, out var secondValue))
  49:         {
  50:             Entity firstEntity = first.Entity;
  51:             var secondRows = firstRows;
  52:             while (secondRows.MoveNext(out var second, out var startValue, out var lastValue))
  53:             {
  54:                 Entity secondEntity = default;
  55:                 bool ascending = false;
  56:                 ComponentReference<TFirst> relationFirst = firstValue;
  57:                 ComponentReference<TSecond> relationSecond = lastValue;
  58:                 if (typeof(TFirst) != typeof(TSecond) || !TRelation.IsSymmetric)
  59:                 {
  60:                     secondEntity = second.Entity;
  61:                     ascending = EntityOrder.Precedes(firstEntity, secondEntity);
  62:                     relationFirst = ascending ? firstValue : startValue;
  63:                     relationSecond = ascending ? lastValue : secondValue;
  64:                 }
  65:                 if (TSelection.RequiresMatch)
  66:                 {
  67:                     if (!TPresence.IsRequired && (!relationFirst.HasValue || !relationSecond.HasValue))
  68:                         continue;
  69:                     if (!predicate.Matches(in relationFirst.Value, in relationSecond.Value))
  70:                         continue;
  71:                 }
  72:                 else if ((TPresence.IsRequired || (relationFirst.HasValue && relationSecond.HasValue)) &&
  73:                     predicate.Matches(in relationFirst.Value, in relationSecond.Value))
  74:                     continue;
  75:                 if (typeof(TFirst) == typeof(TSecond) && TRelation.IsSymmetric)
  76:                 {
  77:                     secondEntity = second.Entity;
  78:                     ascending = EntityOrder.Precedes(firstEntity, secondEntity);
  79:                 }
  80:                 var pair = ascending ? (firstEntity, secondEntity) : (secondEntity, firstEntity);
  81:                 bool continued = TOutput.ReadsComponentRows
  82:                     ? ascending ? output.Add(in pair, first, second) : output.Add(in pair, second, first)
  83:                     : output.Add(in pair);
  84:                 if (!continued)
  85:                     return false;
  86:             }
  87:         }
  88:         return true;
  89:     }
  90: }

  22:     private static bool Enumerate<TOutput>(World world, QueryHandle members, bool exclude, ref TOutput output)
  23:         where TOutput : struct, IRelationOutput<(Entity, Entity)>, allows ref struct
  24:     {
  25:         QueryDefinition original = world.GetQueryDefinition(members);
  26:         QueryDefinition definition = original.WithOptionalRead<TComponent>();
  27:         bool temporaryQuery = definition.CanWrite || !ReferenceEquals(original, definition);
  28:         QueryHandle query = temporaryQuery ? world.Query(definition) : members;
  29:         var execution = new MemberExecution<TOutput>(world, exclude, output);

  44:         where TOutput : struct, IRelationOutput<(Entity, Entity)>, allows ref struct

  47:         private readonly bool _exclude;
  48:
  49:         public MemberExecution(World world, bool exclude, TOutput output)
  50:         {
  51:             _world = world;
  52:             _exclude = exclude;
  53:             Output = output;
  54:         }

  57:         public bool Completed { get; private set; }
  58:
  59:         public void Execute(scoped QueryCursor members)
  60:         {
  61:             var keys = new HashSet<TKey>();
  62:             var missing = new List<Entity>();
  63:             foreach (var member in members.Rows)
  64:             {
  65:                 if (member.Has<TComponent>())
  66:                     keys.Add(member.ReadRef<TComponent>().GetKey());
  67:                 else
  68:                     missing.Add(member.Entity);
  69:             }

  72:             bool firstMissing = false;
