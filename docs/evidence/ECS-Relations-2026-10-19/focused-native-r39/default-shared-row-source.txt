   6: /// <summary>Member-row traversal with callback-scoped optional component columns.</summary>
   7: public ref struct ComponentRowEnumerator<TFirst, TSecond>
   8:     where TFirst : struct
   9:     where TSecond : struct
  10: {
  11:     private QueryRowEnumerator<NoSharedFilter> _rows;
  12:     private ReadOnlySpan<TFirst> _firstComponents;
  13:     private ReadOnlySpan<TSecond> _secondComponents;
  14: 
  15:     internal ComponentRowEnumerator(QueryRowEnumerator<NoSharedFilter> rows)
  16:     {
  17:         _rows = rows;
  18:         _firstComponents = default;
  19:         _secondComponents = default;
  20:     }
  21: 
  22:     [MethodImpl(MethodImplOptions.AggressiveInlining)]
  23:     public bool MoveNext(out QueryRow row, out ComponentReference<TFirst> first, out ComponentReference<TSecond> second)
  24:     {
  25:         Chunk? previous = _rows.CurrentChunk;
  26:         if (!_rows.MoveNext(out row))
  27:         {
  28:             first = default;
  29:             second = default;
  30:             return false;
  31:         }
  32:         if (!ReferenceEquals(previous, _rows.CurrentChunk))
  33:             BindColumns(_rows.CurrentMatch, _rows.CurrentChunk!);
  34: 
  35:         first = _firstComponents.IsEmpty ? default : new(in _firstComponents[row.RowIndex]);
  36:         // Repeated positions borrow the same component type, which proves this reference cast.
  37:         second = typeof(TFirst) == typeof(TSecond)
  38:             ? new(in Unsafe.As<TFirst, TSecond>(ref Unsafe.AsRef(in first.Value)))
  39:             : _secondComponents.IsEmpty ? default : new(in _secondComponents[row.RowIndex]);
  40:         return true;
  41:     }
  42: 
  43:     [MethodImpl(MethodImplOptions.NoInlining)]
  44:     private void BindColumns(QueryArchetypeMatch match, Chunk chunk)
  45:     {
  46:         _firstComponents = QueryAccessGuards.TryRequireReadAccess<TFirst>(match, out int firstColumn)
  47:             ? chunk.ComponentRows<TFirst>(firstColumn) : default;
  48:         if (typeof(TFirst) != typeof(TSecond))
  49:             _secondComponents = QueryAccessGuards.TryRequireReadAccess<TSecond>(match, out int secondColumn)
  50:                 ? chunk.ComponentRows<TSecond>(secondColumn) : default;
  51:     }
  52: }

  43:     public T Read<T>() where T : struct
  44:     {
  45:         int column = QueryAccessGuards.RequireAccess<T>(_match, read: true, write: false);
  46:         return _chunk.ReadComponent<T>(column, _row);
  47:     }

  49:     public ref readonly T ReadRef<T>() where T : struct
  50:     {
  51:         int column = QueryAccessGuards.RequireAccess<T>(_match, read: true, write: false);
  52:         return ref _chunk.ComponentRows<T>(column)[_row];
  53:     }
  54: 
  55:     /// <summary>Finds one present readable component and borrows its value without copying it.</summary>
  56:     [MethodImpl(MethodImplOptions.AggressiveInlining)]
  57:     public bool TryReadRef<T>(out ComponentReference<T> component) where T : struct
  58:     {
  59:         if (QueryAccessGuards.TryRequireReadAccess<T>(_match, out int column))
  60:         {
  61:             component = new(in _chunk.ComponentRows<T>(column)[_row]);
  62:             return true;
  63:         }
  64:         component = default;
  65:         return false;
  66:     }
