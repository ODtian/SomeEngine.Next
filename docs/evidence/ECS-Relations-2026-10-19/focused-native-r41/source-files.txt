   7: /// <summary>Within-query pairing that binds its existing component rows before emitting identities.</summary>
   8: public readonly struct ComponentPairsWithin<TRelation, TFirst, TSecond, TSelection>(
   9:     EntityPairsWithin source, TRelation relation) : IQueryRelationEnumerable<(Entity, Entity)>
  10:     where TRelation : struct, IRelation<TFirst, TSecond>
  11:     where TFirst : struct
  12:     where TSecond : struct
  13:     where TSelection : struct, IRelationSelection
  14: {
  15:     private readonly bool _componentsRequired = ReadComponentRequirements(source);
  16: 
  17:     public World World => source.World;
  18:     public int QueryCount => 1;
  19:     public static bool ReadsFromSingleQuery => true;
  20:     public QueryHandle GetQuery(int position) => source.GetQuery(position);
  21:     public QueryDefinition AddReads(QueryDefinition definition, int position)
  22:     {
  23:         QueryDefinition reads = definition.WithOptionalRead<TFirst>();
  24:         return typeof(TFirst) == typeof(TSecond) ? reads : reads.WithOptionalRead<TSecond>();
  25:     }
  26: 
  27:     [MethodImpl(MethodImplOptions.AggressiveInlining)]
  28:     public bool Enumerate<TOutput>(scoped RelationQueryViews queries, ref TOutput output)
  29:         where TOutput : struct, IQueryRelationOutput<(Entity, Entity)>, allows ref struct
  30:     {
  31:         QueryCursor members = queries.First;
  32:         return _componentsRequired ? Enumerate<RequiredRelationComponents, TOutput>(members, relation, ref output)
  33:             : Enumerate<OptionalRelationComponents, TOutput>(members, relation, ref output);
  34:     }
  35: 
  36:     private static bool ReadComponentRequirements(EntityPairsWithin source)
  37:     {
  38:         QueryDefinition definition = source.World.GetQueryDefinition(source.GetQuery(0));
  39:         return definition.Requires<TFirst>() &&
  40:             (typeof(TFirst) == typeof(TSecond) || definition.Requires<TSecond>());
  41:     }
  42: 
  43:     private static bool Enumerate<TPresence, TOutput>(scoped QueryCursor members, TRelation predicate, ref TOutput output)
  44:         where TPresence : struct, IRelationComponentPresence
  45:         where TOutput : struct, IQueryRelationOutput<(Entity, Entity)>, allows ref struct
  46:     {
  47:         var firstRows = members.ReadRows<TFirst, TSecond>();
  48:         while (firstRows.MoveNext(out var first, out var firstValue, out var secondValue))
  49:         {
  50:             Entity firstEntity = first.Entity;
  51:             var secondRows = firstRows;
  52:             while (secondRows.MoveNext(out var second, out var startValue, out var lastValue))
  53:             {
  54:                 Entity secondEntity = default;
  55:                 bool ascending = false;
  56:                 ComponentReference<TFirst> relationFirst = firstValue;
  57:                 ComponentReference<TSecond> relationSecond = lastValue;
  58:                 if (typeof(TFirst) != typeof(TSecond) || !TRelation.IsSymmetric)
  59:                 {
  60:                     secondEntity = second.Entity;
  61:                     ascending = EntityOrder.Precedes(firstEntity, secondEntity);
  62:                     relationFirst = ascending ? firstValue : startValue;
  63:                     relationSecond = ascending ? lastValue : secondValue;
  64:                 }
  65:                 if (TSelection.RequiresMatch)
  66:                 {
  67:                     if (!TPresence.IsRequired && (!relationFirst.HasValue || !relationSecond.HasValue))
  68:                         continue;
  69:                     if (!predicate.Matches(in relationFirst.Value, in relationSecond.Value))
  70:                         continue;
  71:                 }
  72:                 else if ((TPresence.IsRequired || (relationFirst.HasValue && relationSecond.HasValue)) &&
  73:                     predicate.Matches(in relationFirst.Value, in relationSecond.Value))
  74:                     continue;
  75:                 if (typeof(TFirst) == typeof(TSecond) && TRelation.IsSymmetric)
  76:                 {
  77:                     secondEntity = second.Entity;
  78:                     ascending = EntityOrder.Precedes(firstEntity, secondEntity);
  79:                 }
  80:                 var pair = ascending ? (firstEntity, secondEntity) : (secondEntity, firstEntity);
  81:                 bool continued = TOutput.ReadsComponentRows
  82:                     ? ascending ? output.Add(in pair, first, second) : output.Add(in pair, second, first)
  83:                     : output.Add(in pair);
  84:                 if (!continued)
  85:                     return false;
  86:             }
  87:         }
  88:         return true;
  89:     }
  90: }

  46:     {

  48:         if (TPairing.QueryCount == 1 && TPairing.ReadsFromSingleQuery)
  49:             return Enumerate(World, _exclude, in _relation, in _implementation, first, first, first, ref output);
  50:         QueryCursor second = queries.Second;
  51:         QueryCursor access = queries.Access;
  52:         return Enumerate(World, _exclude, in _relation, in _implementation, first, second, access, ref output);

  55:     private static bool Enumerate<TOutput>(World world, bool exclude, scoped in TRelation relation,
  56:         scoped in TImplementation implementation, scoped QueryCursor first, scoped QueryCursor second,
  57:         scoped QueryCursor access, ref TOutput output)
  58:         where TOutput : struct, IQueryRelationOutput<(Entity, Entity)>, allows ref struct
  59:     {
  60:         var firstGroups = TPairing.ReadsFromSingleQuery ? new KeyGroups<TComponent, TKey>(world, in first)
  61:             : new KeyGroups<TComponent, TKey>(world, in first, in access);
  62:         var secondGroups = TPairing.QueryCount == 1 ? firstGroups : new KeyGroups<TComponent, TKey>(world, in second, in access);
  63:         var firstValues = firstGroups.GetEnumerator();
  64:         while (firstValues.MoveNext(out var firstGroup))
  65:         {
  66:             var secondValues = secondGroups.GetEnumerator();
  67:             while (secondValues.MoveNext(out var secondGroup))
  68:             {
  69:                 bool hasKeys = firstGroup.HasKey && secondGroup.HasKey;
  70:                 bool forwardMatch = hasKeys && implementation.Matches(in relation, in firstGroup.Key, in secondGroup.Key);
  71:                 bool reverseMatch = TPairing.ReordersPairs
  72:                     ? hasKeys && implementation.Matches(in relation, in secondGroup.Key, in firstGroup.Key)
  73:                     : forwardMatch;

  75:                 bool reverseAccepted = reverseMatch != exclude;
  76:                 if (!forwardAccepted && !reverseAccepted)
  77:                     continue;

  83:                         : first.Contains(firstEntity);

  85:                         continue;
  86:                     foreach (Entity secondEntity in secondGroup.Entities)
  87:                     {
  88:                         if (!TPairing.TryPair(in first, in second, firstEntity, secondEntity, out var pair))
  89:                             continue;
  90:                         scoped QueryRow secondRow = default;
  91:                         bool secondMember;
  92:                         if (TPairing.ReadsFromSingleQuery && TOutput.ReadsComponentRows)
  93:                             secondMember = first.TryGetRow(secondEntity, out secondRow);
  94:                         else

  96:                                 ? first.Contains(secondEntity) : second.Contains(secondEntity);
  97:                         if (!secondMember)
  98:                             continue;
  99:                         bool accepted = TPairing.ReordersPairs && pair.Item1 != firstEntity
 100:                             ? reverseAccepted : forwardAccepted;
 101:                         if (!accepted)
 102:                             continue;
 103:                         bool continued = TPairing.ReadsFromSingleQuery && TOutput.ReadsComponentRows
 104:                             ? output.Add(in pair, firstRow, secondRow) : output.Add(in pair);

 106:                             return false;

 108:                 }

 112:     }

   7: /// <summary>Member-row traversal with callback-scoped optional component columns.</summary>
   8: public ref struct ComponentRowEnumerator<TFirst, TSecond>
   9:     where TFirst : struct
  10:     where TSecond : struct
  11: {
  12:     private QueryRowEnumerator<NoSharedFilter> _rows;
  13:     private ReadOnlySpan<TFirst> _firstComponents;
  14:     private ReadOnlySpan<TSecond> _secondComponents;
  15: 
  16:     internal ComponentRowEnumerator(QueryRowEnumerator<NoSharedFilter> rows)
  17:     {
  18:         _rows = rows;
  19:         _firstComponents = default;
  20:         _secondComponents = default;
  21:     }
  22: 
  23:     [MethodImpl(MethodImplOptions.AggressiveInlining)]
  24:     public bool MoveNext(out QueryRow row, out ComponentReference<TFirst> first, out ComponentReference<TSecond> second)
  25:     {
  26:         Chunk? previous = _rows.CurrentChunk;
  27:         if (!_rows.MoveNext(out row))
  28:         {
  29:             first = default;
  30:             second = default;
  31:             return false;
  32:         }
  33:         if (!ReferenceEquals(previous, _rows.CurrentChunk))
  34:             BindColumns(_rows.CurrentMatch, _rows.CurrentChunk!);
  35: 
  36:         // RowIndex and both nonempty columns come from the same chunk.
  37:         first = _firstComponents.IsEmpty ? default
  38:             : new(in Unsafe.Add(ref MemoryMarshal.GetReference(_firstComponents), row.RowIndex));
  39:         // Repeated positions borrow the same component type, which proves this reference cast.
  40:         second = typeof(TFirst) == typeof(TSecond)
  41:             ? new(in Unsafe.As<TFirst, TSecond>(ref Unsafe.AsRef(in first.Value)))
  42:             : _secondComponents.IsEmpty ? default
  43:                 : new(in Unsafe.Add(ref MemoryMarshal.GetReference(_secondComponents), row.RowIndex));
  44:         return true;
  45:     }
  46: 
  47:     [MethodImpl(MethodImplOptions.NoInlining)]
  48:     private void BindColumns(QueryArchetypeMatch match, Chunk chunk)
  49:     {
  50:         _firstComponents = QueryAccessGuards.TryRequireReadAccess<TFirst>(match, out int firstColumn)
  51:             ? chunk.ComponentRows<TFirst>(firstColumn) : default;
  52:         if (typeof(TFirst) != typeof(TSecond))
  53:             _secondComponents = QueryAccessGuards.TryRequireReadAccess<TSecond>(match, out int secondColumn)
  54:                 ? chunk.ComponentRows<TSecond>(secondColumn) : default;
  55:     }
  56: }

 111:     internal void Fill<TEnumerable>(TEnumerable source, scoped RelationQueryViews queries, bool parallel)
 112:         where TEnumerable : struct, ICollisionPartitionEnumerable
 113:     {
 114:         source.Prepare(queries);
 115:         int partitionCount = source.Tree.CreatePartitions(_partitions);
 116:         PartitionCount = partitionCount;
 117:         for (int index = 0; index < partitionCount; index++)
 118:             _countAccesses[index] = JobResourceAccess.Write(_countResource, index, 1);
 119:         var countJob = new CountJob<TEnumerable>(source, this);
 120:         if (parallel)
 121:             ReadOnlyQueryPairJobs.ExecuteParallel(queries.Access, queries.First, in countJob,
 122:                 partitionCount, _countAccesses.AsSpan(0, partitionCount), 1);
 123:         else
 124:             ReadOnlyQueryPairJobs.ExecuteInline(queries.Access, queries.First, in countJob, partitionCount);
 125: 
 126:         BuildOffsets(partitionCount);
 127: 
 128:         int fillCount = 0;
 129:         for (int index = 0; index < partitionCount; index++)
 130:         {
 131:             int count = _offsets[index + 1] - _offsets[index];
 132:             if (count == 0)

 134:             _fillPartitions[fillCount] = index;
